# Trap, Drill and Phonk
# Hats use r (two 32nds), t (three) and q (four) for rolls.
# Rows with 24 steps per bar are 16th-note triplets.
genre: Trap
defaults: bpm=140 time=4/4

  CH  xxxx xxxx xxxx xxrr
  CL  ---- ---- X--- ----
  BD  x--- ---- --x- ----
